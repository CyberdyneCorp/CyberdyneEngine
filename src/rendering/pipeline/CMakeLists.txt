# src/rendering/pipeline/ — layer 4: THE SHADER AND PIPELINE LAYER ABOVE THE FRAME. M8.c task 1b.1.
#
# M8.b's closing gate, verbatim:
#
#   > `FrameAssembly` hands each pass's record callback to its caller and the vertical slice supplies
#   > none, which is why the slice's picture is DRAWN rather than CAPTURED. A person could build all
#   > of M8.b and still not *see* it without writing their own renderer.
#
# This module is the layer that was missing: pipeline state objects, the descriptor sets on the
# engine's own set convention, the per-frame ring, and the record callbacks `FrameSinks::passes`
# takes. It is the FIRST module in the tree that binds a pipeline the engine itself created — every
# earlier one either declared a pass and left it empty (src/rendering/assembly/) or built a renderer
# of its own beside the engine (samples/03-first-light).
#
# WHY IT IS ABOVE assembly/ AND NOT INSIDE IT. `FrameAssembly` states outright that it "does not own
# the shaders or the pipelines" and that "a pass's record callback is the CALLER's". Folding this in
# would make the assembly own a shading model, which is the thing that stops a renderer's structure
# from changing without every material changing with it. What this module is instead is the FIRST
# caller — and `samples/08-vertical-slice` is the second.
#
# THE SPIR-V IS COMMITTED, NOT COMPILED HERE. `shader-system`: "a shipping build SHALL contain
# compiled backend-native shader artefacts and no Slang compiler", and CY_SHADER_SLANG is off by
# default. shaders/frame_spirv.h is generated by shaders/embed_spirv.py from
# src/rendering/shaders/cy/frame.slang and cy/fullscreen.slang, whose header comments carry the
# invocations. A module that only drew on a machine with a shader toolchain would be a module the
# artefact could not use.
#
# Sources are listed explicitly and never globbed, as everywhere else in this tree.
cy_add_module(
    NAME cy_rendering_pipeline
    LAYER rendering
    ALIAS cy::rendering-pipeline
    TYPE STATIC
    SOURCES
        src/frame_pipelines.cpp
        src/frame_bindings.cpp
        src/frame_recorder.cpp
    PUBLIC_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include"
    # shaders/ holds the committed SPIR-V header and nothing else; it is private because a consumer
    # binds a pipeline rather than a shader module.
    PRIVATE_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/shaders"
    PUBLIC_DEPENDENCIES
        cy::core-base
        cy::core-math
        cy::core-memory
        cy::rendering-assembly
        cy::rendering-forward
        cy::rendering-graph
        cy::rendering-lighting
        cy::rendering-material
        cy::rhi
        cy::servers-render)

# The Slang the committed SPIR-V came from lives in src/rendering/shaders/, so this target depends on
# that one: the staged tree must exist for anybody regenerating, and the dependency is what makes
# "which Slang produced this header" answerable from the build graph rather than from a comment.
add_dependencies(cy_rendering_pipeline cy_shaders_stage)

if(CY_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
