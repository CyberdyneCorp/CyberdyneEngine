# src/ui/text/ — an element's text, measured and painted into CyberUI's primitive stream.
#
# `cy::ui` measures and paints content through two interfaces it does not implement,
# `ContentMeasurer` and `ContentPainter`, so that it links no font server. This target is their
# implementation over `cy::servers-text`: a label's text shaped and laid out by the text server and
# drawn as glyph primitives that sample its glyph atlases. It carries two fonts: the interface font
# (interface_font.h, Noto Sans as a distance field, with CY_TEXT), and the built-in public-domain
# bitmap font (builtin_font.h) that #102 drew with, which stays as the fallback face and as the whole
# of the interface's text with CY_TEXT off. src/builtin_font_data.h is GENERATED by
# tools/make_builtin_font.py.
#
# LAYER. `scene` (4), beside cy::ui.
cy_add_module(
    NAME cy_ui_text
    LAYER scene
    ALIAS cy::ui-text
    TYPE STATIC
    SOURCES
        src/builtin_font.cpp
        src/interface_font.cpp
        src/text_painter.cpp
    PUBLIC_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PUBLIC_DEPENDENCIES cy::ui cy::servers-text)

# The interface font (interface_font.h): deps/fonts/NotoSans-Latin-VF.ttf compiled in, when the
# complete text backend that can draw it is. With CY_TEXT off the font is not carried and the
# built-in grid font is the interface's only face, exactly as before.
if(CY_TEXT)
    include(embed)
    cy_embed_file(cy_ui_text
        INPUT "${CMAKE_SOURCE_DIR}/deps/fonts/NotoSans-Latin-VF.ttf"
        SYMBOL interface_font
        NAMESPACE cy::ui::embedded)
endif()

if(CY_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
