# src/ui/text/ — an element's text, measured and painted into CyberUI's primitive stream.
#
# `cy::ui` measures and paints content through two interfaces it does not implement,
# `ContentMeasurer` and `ContentPainter`, so that it links no font server. This target is their
# implementation over `cy::servers-text`: a label's text shaped and laid out by the text server and
# drawn as glyph primitives that sample its glyph atlas. It also carries the built-in interface font
# (builtin_font.h), a public-domain bitmap font that stands in until the engine imports real fonts
# (`text-and-fonts`, issue #86). src/builtin_font_data.h is GENERATED by
# tools/make_builtin_font.py.
#
# LAYER. `scene` (4), beside cy::ui.
cy_add_module(
    NAME cy_ui_text
    LAYER scene
    ALIAS cy::ui-text
    TYPE STATIC
    SOURCES
        src/builtin_font.cpp
        src/text_painter.cpp
    PUBLIC_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PUBLIC_DEPENDENCIES cy::ui cy::servers-text)

if(CY_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
